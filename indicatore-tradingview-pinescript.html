Humble Trading — Prop Firm Shield & ICT Killzones [v5]

L'indicatore gratuito per TradingView progettato per chi opera su conti Prop Firm (XAUUSD, EURUSD, NAS100, MNQ): disegna in automatico i massimi e minimi di Londra, evidenzia gli Sweep di Liquidità delle 14:30 CET e calcola in tempo reale la size anti-bruciatura direttamente sul grafico.

🛡️ Dashboard Lotti & Drawdown sul Grafico

Mostra nell'angolo in alto a destra la size esatta in Lotti o Contratti per rischiare lo 0,40% e quanti stop consecutivi hai di margine prima del 5% Daily Drawdown.

🎯 Massimi & Minimi di Londra Automatici

Traccia automaticamente il London High e London Low (08:00 - 11:00 ora italiana) come bersagli di liquidità per la sessione di New York.

⚡ Rilevatore NY Sweep (14:30 - 16:00 CET) + FVG

Segnala con un triangolo sul grafico quando il prezzo spazza la liquidità di Londra e forma un Fair Value Gap (FVG) istituzionale, più l'allerta Rollover 00:00 CET.

📋 Codice Sorgente PineScript v5 (Pronto da Copiare)

Apri TradingView → scheda "Pine Editor" in basso → Incolla il codice → Clicca "Aggiungi al grafico".

// © Humble Trading Community (https://t.me/Hubmles | https://discord.gg/VFt4y4MY5)
//@version=5
indicator("Humble Trading — Prop Firm Shield & ICT Killzones [v5]", shorttitle="HT Prop Shield & ICT", overlay=true, max_lines_count=500, max_boxes_count=500, max_labels_count=500)

grp_prop = "🛡️ Configurazione Conto Prop Firm (Humble Trading)"
acc_balance = input.float(100000, "Capitale Conto Prop ($)", minval=1000, step=5000, group=grp_prop)
risk_pct    = input.float(0.40, "Rischio per Trade (%)", minval=0.10, maxval=3.0, step=0.05, group=grp_prop)
daily_dd    = input.float(5.0, "Limite Daily Drawdown (%)", minval=1.0, maxval=10.0, step=0.5, group=grp_prop)
sl_points   = input.float(25.0, "Stop Loss Pianificato (Punti XAUUSD / Pips)", minval=1.0, step=1.0, group=grp_prop)
asset_type  = input.string("XAUUSD (Oro CFD)", "Tipo Strumento", options=["XAUUSD (Oro CFD)", "Forex (EURUSD/GBPUSD)", "NAS100 CFD", "MNQ (Micro Nasdaq Futures)"], group=grp_prop)

grp_ict = "⚡ Sessioni ICT & Liquidity Sweep (Ora Italiana Europe/Rome)"
show_lon    = input.bool(true, "Mostra Massimo/Minimo Sessione Londra (08:00-11:00 CET)", group=grp_ict)
show_ny_kz  = input.bool(true, "Evidenzia Finestra NY Liquidity Sweep (14:30-16:00 CET)", group=grp_ict)
show_fvg    = input.bool(true, "Evidenzia Fair Value Gap (FVG) Istituzionali", group=grp_ict)

in_london = not na(time(timeframe.period, "0800-1100", "Europe/Rome"))
in_ny_kz  = not na(time(timeframe.period, "1430-1600", "Europe/Rome"))
in_roll   = not na(time(timeframe.period, "2345-0100", "Europe/Rome"))

var float lon_high = na
var float lon_low  = na

if in_london and not in_london[1]
    lon_high := high
    lon_low  := low
else if in_london
    lon_high := math.max(nz(lon_high, high), high)
    lon_low  := math.min(nz(lon_low, low), low)

plot(show_lon and not in_london ? lon_high : na, "London High (Buy-Side Liquidity)", color=color.new(#F59E0B, 15), style=plot.style_linebr, linewidth=2)
plot(show_lon and not in_london ? lon_low  : na, "London Low (Sell-Side Liquidity)", color=color.new(#10B981, 15), style=plot.style_linebr, linewidth=2)

bgcolor(show_ny_kz and in_ny_kz ? color.new(#F59E0B, 92) : na, title="Finestra NY Sweep 14:30-16:00 CET")
bgcolor(in_roll ? color.new(#EF4444, 90) : na, title="Allerta Rollover Spread 00:00 CET")

bearish_sweep = in_ny_kz and not na(lon_high) and high > lon_high and close < lon_high
bullish_sweep = in_ny_kz and not na(lon_low)  and low < lon_low   and close > lon_low

plotshape(bearish_sweep, title="Bearish Liquidity Sweep (London High)", style=shape.triangledown, location=location.abovebar, color=#EF4444, size=size.small, text="SWEEP HIGH")
plotshape(bullish_sweep, title="Bullish Liquidity Sweep (London Low)",  style=shape.triangleup,   location=location.belowbar, color=#10B981, size=size.small, text="SWEEP LOW")

bull_fvg = show_fvg and low > high[2] and close[1] > high[2]
bear_fvg = show_fvg and high < low[2] and close[1] < low[2]

if bull_fvg and in_ny_kz
    box.new(bar_index[2], low, bar_index + 4, high[2], border_color=color.new(#10B981, 50), bgcolor=color.new(#10B981, 85))

if bear_fvg and in_ny_kz
    box.new(bar_index[2], low[2], bar_index + 4, high, border_color=color.new(#EF4444, 50), bgcolor=color.new(#EF4444, 85))

risk_usd = acc_balance * (risk_pct / 100.0)
val_per_pt = asset_type == "XAUUSD (Oro CFD)" ? 100.0 : asset_type == "Forex (EURUSD/GBPUSD)" ? 10.0 : asset_type == "NAS100 CFD" ? 1.0 : 2.0
raw_lots = risk_usd / (sl_points * val_per_pt)
safe_lots = math.floor(raw_lots * 100.0) / 100.0
consecutive_stops = daily_dd / risk_pct

var table hud = table.new(position.top_right, 2, 7, bgcolor=color.new(#070A10, 10), border_color=color.new(#F59E0B, 50), border_width=1)

if barstate.islast
    table.cell(hud, 0, 0, "⚡ HUMBLE TRADING SHIELD", text_color=#F59E0B, text_size=size.small, bgcolor=color.new(#14161F, 0))
    table.cell(hud, 1, 0, "t.me/Hubmles", text_color=#38BDF8, text_size=size.small, bgcolor=color.new(#14161F, 0))
    table.cell(hud, 0, 1, "Conto Prop / Rischio", text_color=color.gray, text_size=size.tiny)
    table.cell(hud, 1, 1, "$" + str.tostring(acc_balance, "#,###") + " (" + str.tostring(risk_pct, "#.##") + "% = $" + str.tostring(risk_usd, "#") + ")", text_color=color.white, text_size=size.tiny)
    table.cell(hud, 0, 2, "SIZE CONSIGLIATA", text_color=#10B981, text_size=size.small)
    table.cell(hud, 1, 2, str.tostring(safe_lots, "#.##") + (asset_type == "MNQ (Micro Nasdaq Futures)" ? " Contratti" : " Lotti"), text_color=#F59E0B, text_size=size.small)
    table.cell(hud, 0, 3, "Buffer Daily DD (" + str.tostring(daily_dd, "#.#") + "%)", text_color=color.gray, text_size=size.tiny)
    table.cell(hud, 1, 3, str.tostring(consecutive_stops, "#.#") + " Stop Consecutivi", text_color=#10B981, text_size=size.tiny)
    table.cell(hud, 0, 4, "London High / Low", text_color=color.gray, text_size=size.tiny)
    table.cell(hud, 1, 4, str.tostring(lon_high, "#.##") + " / " + str.tostring(lon_low, "#.##"), text_color=color.white, text_size=size.tiny)
    table.cell(hud, 0, 5, "Top Prop 2026 CFD / Fut", text_color=color.gray, text_size=size.tiny)
    table.cell(hud, 1, 5, "Propr.xyz · Breakout · Tradeify", text_color=#F59E0B, text_size=size.tiny)
    table.cell(hud, 0, 6, "Community Gratuita", text_color=color.gray, text_size=size.tiny)
    table.cell(hud, 1, 6, "discord.gg/VFt4y4MY5", text_color=#38BDF8, text_size=size.tiny)
